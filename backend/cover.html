// errUserNotActive is a sentinel error returned by checkUserActive when the
// user is not active. The HTTP response has already been written at that point.
var errUserNotActive = errors.New("user not active")

// errNoExchangeRates is a sentinel error indicating the exchange_rates table is empty.
var errNoExchangeRates = errors.New("exchange rates table is empty")

// checkUserActive verifies that the user exists, is active, and is not soft-deleted.
// Returns errUserNotActive (after writing a 401 JSON response) when the user
// cannot be found, is deactivated, or is deleted. Returns nil when the user is active.
func (h *Handler) checkUserActive(c echo.Context, userID int) error {
        user, err := h.userRepo.GetByID(userID)
        if err != nil {
                h.logger.Error("failed to get user", "error", err)
                _ = c.JSON(http.StatusUnauthorized, ErrorResponse{Detail: "User not activated"})
                return errUserNotActive
        }
        if !user.IsActive || user.IsDeleted {
                _ = c.JSON(http.StatusUnauthorized, ErrorResponse{Detail: "User not activated"})
                return errUserNotActive
        }
        return nil
}

type Handler struct {
        plannedTxRepo    repositories.PlannedTransactionRepository
        accountRepo      repositories.AccountRepository
        currencyRepo     repositories.CurrencyRepository
        exchangeRateRepo repositories.ExchangeRateRepository
        userRepo         repositories.UserRepository
        logger           *slog.Logger
}

func New(
        plannedTxRepo repositories.PlannedTransactionRepository,
        accountRepo repositories.AccountRepository,
        currencyRepo repositories.CurrencyRepository,
        exchangeRateRepo repositories.ExchangeRateRepository,
        userRepo repositories.UserRepository,
        logger *slog.Logger,
) *Handler {
        return &Handler{
                plannedTxRepo:    plannedTxRepo,
                accountRepo:      accountRepo,
                currencyRepo:     currencyRepo,
                exchangeRateRepo: exchangeRateRepo,
                userRepo:         userRepo,
                logger:           logger,
        }
}

func (h *Handler) RegisterRoutes(g *echo.Group) {
        g.POST("/future-balance", h.CalculateFutureBalance)
        g.POST("/projection", h.GetBalanceProjection)
}

// rateCache caches exchange rate lookups per date within a single request.
type rateCache struct {
        cache map[string]*repositories.ExchangeRateListResponse
        repo  repositories.ExchangeRateRepository
}

func newRateCache(repo repositories.ExchangeRateRepository) *rateCache {
        return &rateCache{
                cache: make(map[string]*repositories.ExchangeRateListResponse),
                repo:  repo,
        }
}

func (rc *rateCache) getRatesForDate(dateStr string) (*repositories.ExchangeRateListResponse, error) {
        if cached, ok := rc.cache[dateStr]; ok {
                return cached, nil
        }
        rates, err := rc.repo.GetAllRatesForDate(dateStr)
        if err != nil {
                return nil, err
        }
        rc.cache[dateStr] = rates
        return rates, nil
}

// convertToBaseCurrency converts an amount from sourceCurrencyCode to baseCurrencyCode
// using exchange rates for the given date.
//
// Error semantics:
//   - Returns errNoExchangeRates when the exchange_rates table is completely empty (sql.ErrNoRows).
//   - Returns a non-nil error when a specific currency code is missing from the rates map.
//   - Returns nil error on successful conversion.
func (h *Handler) convertToBaseCurrency(
        amount decimal.Decimal,
        sourceCurrencyCode, baseCurrencyCode string,
        dateStr string,
        rc *rateCache,
) (decimal.Decimal, error) {
        if sourceCurrencyCode == baseCurrencyCode {
                return amount, nil
        }

        rates, err := rc.getRatesForDate(dateStr)
        if err != nil {
                if errors.Is(err, sql.ErrNoRows) {
                        return decimal.Zero, errNoExchangeRates
                }
                h.logger.Warn("exchange rates unavailable",
                        "date", dateStr,
                        "source", sourceCurrencyCode,
                        "target", baseCurrencyCode,
                        "error", err,
                )
                return decimal.Zero, fmt.Errorf("exchange rates unavailable for %s: %w", dateStr, err)
        }

        if rates == nil || len(rates.Rates) == 0 {
                return decimal.Zero, errNoExchangeRates
        }

        sourceRate, sourceOK := rates.Rates[sourceCurrencyCode]
        targetRate, targetOK := rates.Rates[baseCurrencyCode]

        if !sourceOK || !targetOK || sourceRate == 0 || targetRate == 0 {
                h.logger.Warn("exchange rate not found for currency pair",
                        "date", dateStr,
                        "source", sourceCurrencyCode,
                        "target", baseCurrencyCode,
                        "source_rate_found", sourceOK,
                        "target_rate_found", targetOK,
                )
                return decimal.Zero, fmt.Errorf("currency rate missing: source=%s(%v) target=%s(%v)",
                        sourceCurrencyCode, sourceOK, baseCurrencyCode, targetOK)
        }

        // converted = amount / sourceRate * targetRate
        sourceRateDec := decimal.NewFromFloat(sourceRate)
        targetRateDec := decimal.NewFromFloat(targetRate)
        return amount.Div(sourceRateDec).Mul(targetRateDec), nil
}

// DTOs
type FutureBalanceRequest struct {
        TargetDate      time.Time `json:"targetDate" validate:"required"`
        AccountIDs      []int     `json:"accountIds"`
        IncludeInactive bool      `json:"includeInactive"`
}

type AccountBalanceProjection struct {
        AccountID            int             `json:"accountId"`
        AccountName          string          `json:"accountName"`
        CurrencyCode         string          `json:"currencyCode"`
        CurrentBalance       decimal.Decimal `json:"currentBalance"`
        ProjectedBalance     decimal.Decimal `json:"projectedBalance"`
        TotalPlannedIncome   decimal.Decimal `json:"totalPlannedIncome"`
        TotalPlannedExpenses decimal.Decimal `json:"totalPlannedExpenses"`
}

type FutureBalanceResponse struct {
        TargetDate            time.Time                  `json:"targetDate"`
        BaseCurrencyCode      string                     `json:"baseCurrencyCode"`
        TotalCurrentBalance   decimal.Decimal            `json:"totalCurrentBalance"`
        TotalProjectedBalance decimal.Decimal            `json:"totalProjectedBalance"`
        TotalPlannedIncome    decimal.Decimal            `json:"totalPlannedIncome"`
        TotalPlannedExpenses  decimal.Decimal            `json:"totalPlannedExpenses"`
        IncomeCount           int                        `json:"incomeCount"`
        ExpensesCount         int                        `json:"expensesCount"`
        Accounts              []AccountBalanceProjection `json:"accounts"`
}

type BalanceProjectionRequest struct {
        StartDate       time.Time `json:"startDate"`
        EndDate         time.Time `json:"endDate" validate:"required"`
        Period          string    `json:"period"` // daily, weekly, monthly
        AccountIDs      []int     `json:"accountIds"`
        IncludeInactive bool      `json:"includeInactive"`
}

type BalanceProjectionPoint struct {
        Date     time.Time       `json:"date"`
        Balance  decimal.Decimal `json:"balance"`
        Income   decimal.Decimal `json:"income"`
        Expenses decimal.Decimal `json:"expenses"`
}

type BalanceProjectionResponse struct {
        StartDate        time.Time                `json:"startDate"`
        EndDate          time.Time                `json:"endDate"`
        Period           string                   `json:"period"`
        BaseCurrencyCode string                   `json:"baseCurrencyCode"`
        ProjectionPoints []BalanceProjectionPoint `json:"projectionPoints"`
}

type ErrorResponse struct {
        Detail string `json:"detail"`
}

func (h *Handler) CalculateFutureBalance(c echo.Context) error {
        userID := c.Get("user_id").(int)
        if err := h.checkUserActive(c, userID); err != nil {
                return nil
        }

        var req FutureBalanceRequest
        if err := c.Bind(&req); err != nil {
                return c.JSON(http.StatusUnprocessableEntity, ErrorResponse{Detail: "Invalid request data"})
        }
        if err := c.Validate(&req); err != nil {
                return c.JSON(http.StatusUnprocessableEntity, ErrorResponse{Detail: "Invalid request data"})
        }

        h.logger.Info("Calculating future balance",
                "user_id", userID,
                "target_date", req.TargetDate,
                "account_ids", req.AccountIDs,
        )

        // Get user's base currency
        user, err := h.userRepo.GetByID(userID)
        if err != nil {
                h.logger.Error("failed to get user settings", "error", err)
                return c.JSON(http.StatusInternalServerError, ErrorResponse{Detail: "Error calculating future balance"})
        }

        baseCurrency, err := h.currencyRepo.GetByID(user.BaseCurrencyID)
        if err != nil {
                h.logger.Error("failed to get base currency", "error", err)
                return c.JSON(http.StatusInternalServerError, ErrorResponse{Detail: "Error calculating future balance"})
        }

        // Get user accounts
        accountFilters := repositories.AccountFilters{
                IncludeHidden:   true,
                IncludeArchived: false,
        }
        accounts, err := h.accountRepo.GetByUserID(userID, accountFilters)
        if err != nil {
                h.logger.Error("failed to get accounts", "error", err)
                return c.JSON(http.StatusInternalServerError, ErrorResponse{Detail: "Error calculating future balance"})
        }

        // Filter by account IDs if provided
        accountIDSet := make(map[int]bool)
        for _, id := range req.AccountIDs {
                accountIDSet[id] = true
        }

        // Calculate days until target date
        days := int(time.Until(req.TargetDate).Hours() / 24)
        if days < 0 {
                days = 0
        }

        // Get upcoming planned transactions
        plannedTxs, err := h.plannedTxRepo.GetUpcomingOccurrences(userID, days, req.IncludeInactive)
        if err != nil {
                h.logger.Error("failed to get planned transactions", "error", err)
                return c.JSON(http.StatusInternalServerError, ErrorResponse{Detail: "Error calculating future balance"})
        }

        // Initialize exchange rate cache for currency conversion
        rc := newRateCache(h.exchangeRateRepo)

        // Aggregate global totals from all planned transactions, converting to base currency
        globalPlannedIncome := decimal.Zero
        globalPlannedExpenses := decimal.Zero
        incomeCount := 0
        expensesCount := 0

        for _, ptx := range plannedTxs {
                // Look up the currency for this planned transaction
                ptxCurrency, currErr := h.currencyRepo.GetByID(ptx.CurrencyID)
                if currErr != nil {
                        h.logger.Warn("failed to get currency for planned tx, skipping",
                                "currency_id", ptx.CurrencyID, "error", currErr)
                        continue
                }

                // Convert to base currency using the occurrence date
                dateStr := ptx.OccurrenceDate.Format("2006-01-02")
                convertedAmount, convErr := h.convertToBaseCurrency(
                        ptx.Amount, ptxCurrency.Code, baseCurrency.Code, dateStr, rc,
                )
                if convErr != nil {
                        if errors.Is(convErr, errNoExchangeRates) {
                                h.logger.Error("exchange rates table is empty, cannot calculate future balance")
                                return c.JSON(http.StatusInternalServerError, ErrorResponse{Detail: "Exchange rates unavailable"})
                        }
                        h.logger.Warn("skipping planned tx due to missing exchange rate",
                                "planned_tx_id", ptx.PlannedTransactionID, "error", convErr)
                        continue
                }

                if ptx.IsIncome {
                        globalPlannedIncome = globalPlannedIncome.Add(convertedAmount)
                        incomeCount++
                } else {
                        globalPlannedExpenses = globalPlannedExpenses.Add(convertedAmount)
                        expensesCount++
                }
        }

        // Build account projections, converting balances to base currency for the total
        totalCurrentBalance := decimal.Zero
        todayStr := time.Now().Format("2006-01-02")

        var accountProjections []AccountBalanceProjection

        for _, account := range accounts {
                if len(req.AccountIDs) > 0 && !accountIDSet[account.ID] {
                        continue
                }

                currency, currErr := h.currencyRepo.GetByID(account.CurrencyID)
                if currErr != nil {
                        continue
                }

                // Per-account projections show the account's own currency balance (not converted)
                accountProjections = append(accountProjections, AccountBalanceProjection{
                        AccountID:            account.ID,
                        AccountName:          account.Name,
                        CurrencyCode:         currency.Code,
                        CurrentBalance:       account.Balance,
                        ProjectedBalance:     account.Balance,
                        TotalPlannedIncome:   decimal.Zero,
                        TotalPlannedExpenses: decimal.Zero,
                })

                // Convert account balance to base currency for the total
                convertedBalance, convErr := h.convertToBaseCurrency(
                        account.Balance, currency.Code, baseCurrency.Code, todayStr, rc,
                )
                if convErr != nil {
                        if errors.Is(convErr, errNoExchangeRates) {
                                h.logger.Error("exchange rates table is empty, cannot calculate future balance")
                                return c.JSON(http.StatusInternalServerError, ErrorResponse{Detail: "Exchange rates unavailable"})
                        }
                        h.logger.Warn("could not convert account balance to base currency, using raw balance",
                                "account_id", account.ID, "currency", currency.Code, "error", convErr)
                        convertedBalance = account.Balance
                }
                totalCurrentBalance = totalCurrentBalance.Add(convertedBalance)
        }

        totalProjectedBalance := totalCurrentBalance.Add(globalPlannedIncome).Sub(globalPlannedExpenses)

        return c.JSON(http.StatusOK, FutureBalanceResponse{
                TargetDate:            req.TargetDate,
                BaseCurrencyCode:      baseCurrency.Code,
                TotalCurrentBalance:   totalCurrentBalance,
                TotalProjectedBalance: totalProjectedBalance,
                TotalPlannedIncome:    globalPlannedIncome,
                TotalPlannedExpenses:  globalPlannedExpenses,
                IncomeCount:           incomeCount,
                ExpensesCount:         expensesCount,
                Accounts:              accountProjections,
        })
}

// generateDatePoints generates date points for the projection based on the period type.
// Python's algorithm:
//   - Daily: each day from start to end
//   - Weekly: first point = start_date, then find next Monday, generate end-of-week
//     (Sunday) dates until end_date, cap last point at end_date
//   - Monthly: first point = start_date, then find first of next month, generate
//     end-of-month dates until end_date, cap last point at end_date
func generateDatePoints(startDate, endDate time.Time, period string) []time.Time {
        // Normalize dates to start of day (strip time component)
        start := time.Date(startDate.Year(), startDate.Month(), startDate.Day(), 0, 0, 0, 0, startDate.Location())
        end := time.Date(endDate.Year(), endDate.Month(), endDate.Day(), 0, 0, 0, 0, endDate.Location())

        var points []time.Time

        switch period {
        case "weekly":
                // First point is start_date
                points = append(points, start)

                // Find the next Monday after start_date
                daysUntilMonday := (8 - int(start.Weekday())) % 7
                if daysUntilMonday == 0 {
                        daysUntilMonday = 7
                }
                nextMonday := start.AddDate(0, 0, daysUntilMonday)

                // Generate Sunday (end-of-week) dates from that Monday onwards
                for {
                        sunday := nextMonday.AddDate(0, 0, 6) // Sunday = Monday + 6
                        if sunday.After(end) {
                                // Cap at end date if we haven't already added it
                                if !points[len(points)-1].Equal(end) {
                                        points = append(points, end)
                                }
                                break
                        }
                        points = append(points, sunday)
                        nextMonday = nextMonday.AddDate(0, 0, 7)
                }

        case "monthly":
                // First point is start_date
                points = append(points, start)

                // Find the first day of the next month
                firstOfNextMonth := time.Date(start.Year(), start.Month()+1, 1, 0, 0, 0, 0, start.Location())

                // Generate end-of-month dates
                for {
                        // End of this month = first of next month - 1 day
                        endOfMonth := firstOfNextMonth.AddDate(0, 0, -1)
                        if endOfMonth.After(end) {
                                // Cap at end date
                                if !points[len(points)-1].Equal(end) {
                                        points = append(points, end)
                                }
                                break
                        }
                        points = append(points, endOfMonth)
                        // Move to the first of the month after
                        firstOfNextMonth = firstOfNextMonth.AddDate(0, 1, 0)
                }

        default: // daily
                for current := start; !current.After(end); current = current.AddDate(0, 0, 1) {
                        points = append(points, current)
                }
        }

        return points
}

func (h *Handler) GetBalanceProjection(c echo.Context) error {
        userID := c.Get("user_id").(int)
        if err := h.checkUserActive(c, userID); err != nil {
                return nil
        }

        var req BalanceProjectionRequest
        if err := c.Bind(&req); err != nil {
                return c.JSON(http.StatusUnprocessableEntity, ErrorResponse{Detail: "Invalid request data"})
        }
        if err := c.Validate(&req); err != nil {
                return c.JSON(http.StatusUnprocessableEntity, ErrorResponse{Detail: "Invalid request data"})
        }

        // Default start date to now if not provided
        if req.StartDate.IsZero() {
                req.StartDate = time.Now()
        }

        // Default period to daily
        if req.Period == "" {
                req.Period = "daily"
        }

        h.logger.Info("Getting balance projection",
                "user_id", userID,
                "start_date", req.StartDate,
                "end_date", req.EndDate,
                "period", req.Period,
        )

        // Get user's base currency
        user, err := h.userRepo.GetByID(userID)
        if err != nil {
                h.logger.Error("failed to get user settings", "error", err)
                return c.JSON(http.StatusInternalServerError, ErrorResponse{Detail: "Error generating balance projection"})
        }

        baseCurrency, err := h.currencyRepo.GetByID(user.BaseCurrencyID)
        if err != nil {
                h.logger.Error("failed to get base currency", "error", err)
                return c.JSON(http.StatusInternalServerError, ErrorResponse{Detail: "Error generating balance projection"})
        }

        // Get user accounts
        accountFilters := repositories.AccountFilters{
                IncludeHidden:   true,
                IncludeArchived: false,
        }
        accounts, err := h.accountRepo.GetByUserID(userID, accountFilters)
        if err != nil {
                h.logger.Error("failed to get accounts", "error", err)
                return c.JSON(http.StatusInternalServerError, ErrorResponse{Detail: "Error generating balance projection"})
        }

        // Filter by account IDs if provided
        accountIDSet := make(map[int]bool)
        for _, id := range req.AccountIDs {
                accountIDSet[id] = true
        }

        // Initialize exchange rate cache for currency conversion
        rc := newRateCache(h.exchangeRateRepo)
        todayStr := time.Now().Format("2006-01-02")

        // Calculate current balance, converting to base currency
        currentBalance := decimal.Zero
        for _, account := range accounts {
                if len(req.AccountIDs) > 0 && !accountIDSet[account.ID] {
                        continue
                }

                currency, currErr := h.currencyRepo.GetByID(account.CurrencyID)
                if currErr != nil {
                        h.logger.Warn("failed to get currency for account, using raw balance",
                                "account_id", account.ID, "error", currErr)
                        currentBalance = currentBalance.Add(account.Balance)
                        continue
                }

                convertedBalance, convErr := h.convertToBaseCurrency(
                        account.Balance, currency.Code, baseCurrency.Code, todayStr, rc,
                )
                if convErr != nil {
                        if errors.Is(convErr, errNoExchangeRates) {
                                h.logger.Error("exchange rates table is empty, cannot generate projection")
                                return c.JSON(http.StatusInternalServerError, ErrorResponse{Detail: "Exchange rates unavailable"})
                        }
                        h.logger.Warn("could not convert account balance to base currency, using raw balance",
                                "account_id", account.ID, "currency", currency.Code, "error", convErr)
                        currentBalance = currentBalance.Add(account.Balance)
                        continue
                }
                currentBalance = currentBalance.Add(convertedBalance)
        }

        // Calculate days for projection
        days := int(req.EndDate.Sub(req.StartDate).Hours() / 24)
        if days < 0 {
                return c.JSON(http.StatusBadRequest, ErrorResponse{Detail: "End date must be after start date"})
        }

        // Get planned transactions
        plannedTxs, err := h.plannedTxRepo.GetUpcomingOccurrences(userID, days, req.IncludeInactive)
        if err != nil {
                h.logger.Error("failed to get planned transactions", "error", err)
                return c.JSON(http.StatusInternalServerError, ErrorResponse{Detail: "Error generating balance projection"})
        }

        // Generate date points using proper calendar-based algorithm
        datePoints := generateDatePoints(req.StartDate, req.EndDate, req.Period)

        // Generate projection points by iterating over each period
        var projectionPoints []BalanceProjectionPoint
        runningBalance := currentBalance

        for i, point := range datePoints {
                // Determine the period boundaries for collecting transactions
                var periodStart, periodEnd time.Time

                if i == 0 {
                        // First point: period_start = beginning of start_date
                        periodStart = time.Date(point.Year(), point.Month(), point.Day(), 0, 0, 0, 0, point.Location())
                } else {
                        // Subsequent points: period_start = day after previous point
                        prev := datePoints[i-1]
                        periodStart = time.Date(prev.Year(), prev.Month(), prev.Day(), 0, 0, 0, 0, prev.Location()).AddDate(0, 0, 1)
                }
                periodEnd = time.Date(point.Year(), point.Month(), point.Day(), 23, 59, 59, 999999999, point.Location())

                // Collect all planned transaction occurrences within this period
                periodIncome := decimal.Zero
                periodExpenses := decimal.Zero

                for _, ptx := range plannedTxs {
                        occDate := time.Date(ptx.OccurrenceDate.Year(), ptx.OccurrenceDate.Month(), ptx.OccurrenceDate.Day(),
                                0, 0, 0, 0, ptx.OccurrenceDate.Location())

                        if (occDate.Equal(periodStart) || occDate.After(periodStart)) &&
                                (occDate.Equal(periodEnd) || occDate.Before(periodEnd)) {

                                // Convert to base currency using the occurrence date
                                ptxCurrency, currErr := h.currencyRepo.GetByID(ptx.CurrencyID)
                                if currErr != nil {
                                        h.logger.Warn("failed to get currency for planned tx, skipping",
                                                "currency_id", ptx.CurrencyID, "error", currErr)
                                        continue
                                }

                                dateStr := occDate.Format("2006-01-02")
                                convertedAmount, convErr := h.convertToBaseCurrency(
                                        ptx.Amount, ptxCurrency.Code, baseCurrency.Code, dateStr, rc,
                                )
                                if convErr != nil {
                                        if errors.Is(convErr, errNoExchangeRates) {
                                                h.logger.Error("exchange rates table is empty, cannot generate projection")
                                                return c.JSON(http.StatusInternalServerError, ErrorResponse{Detail: "Exchange rates unavailable"})
                                        }
                                        h.logger.Warn("skipping planned tx due to missing exchange rate",
                                                "planned_tx_id", ptx.PlannedTransactionID, "error", convErr)
                                        continue
                                }

                                if ptx.IsIncome {
                                        periodIncome = periodIncome.Add(convertedAmount)
                                } else {
                                        periodExpenses = periodExpenses.Add(convertedAmount)
                                }
                        }
                }

                runningBalance = runningBalance.Add(periodIncome).Sub(periodExpenses)

                projectionPoints = append(projectionPoints, BalanceProjectionPoint{
                        Date:     point,
                        Balance:  runningBalance,
                        Income:   periodIncome,
                        Expenses: periodExpenses,
                })
        }

        return c.JSON(http.StatusOK, BalanceProjectionResponse{
                StartDate:        req.StartDate,
                EndDate:          req.EndDate,
                Period:           req.Period,
                BaseCurrencyCode: baseCurrency.Code,
                ProjectionPoints: projectionPoints,
        })
}
